cmake_minimum_required(VERSION 3.24)
project(hdmi_saved_validation LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(icraft-zg330backend 3.39.0.0 EXACT REQUIRED)
add_executable(hdmi_saved_skeleton saved_skeleton.cpp hdmi_output.cpp hdmi_canvas.cpp skeleton_render.cpp)
target_compile_options(hdmi_saved_skeleton PRIVATE -Wall -Wextra -Werror -ffp-contract=off)
target_link_options(hdmi_saved_skeleton PRIVATE -Wl,--no-as-needed)
target_link_libraries(hdmi_saved_skeleton PRIVATE Icraft::ZG330Backend)
