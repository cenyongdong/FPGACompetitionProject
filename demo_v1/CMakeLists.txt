cmake_minimum_required(VERSION 3.24)
project(pose_demo_v1 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(Threads REQUIRED)
find_package(icraft-hostbackend 3.39.0.0 EXACT REQUIRED)
find_package(icraft-zg330backend 3.39.0.0 EXACT REQUIRED)
set(B ${CMAKE_CURRENT_SOURCE_DIR}/shared_fp16)
option(DEMO_READ_DIAGNOSTIC "Bounded op192 observations; no read/wait substitution" OFF)
option(DEMO_D47_ISOLATION "Static public-SDK read control and Host index oracle; no extra device queries" OFF)
set(READ_BRIDGE ${B}/read_input_bridge.cpp)
set(STAGE_OBSERVATION ${B}/host_stage_observation.cpp)
if(DEMO_READ_DIAGNOSTIC)
    set(READ_BRIDGE diagnostics/read_input_bridge.cpp)
    set(STAGE_OBSERVATION diagnostics/host_stage_observation.cpp)
endif()
add_executable(pose_demo_v1 board/main.cpp output/pose_frame.cpp
    ${B}/application_core.cpp ${READ_BRIDGE} ${B}/input_read_sdk.cpp
    ${B}/input_read_contract.cpp ${B}/host_cpu_adapter.cpp ${B}/preprocess.cpp
    ${B}/host_failure_evidence.cpp ${B}/failure_source_readback.cpp
    ${STAGE_OBSERVATION} ${B}/invocation_evidence.cpp
    ${B}/producer_count_observation.cpp ${B}/tensor_identity_observation.cpp)
target_include_directories(pose_demo_v1 PRIVATE ${B} board output)
if(DEMO_READ_DIAGNOSTIC)
    target_include_directories(pose_demo_v1 BEFORE PRIVATE diagnostics)
    target_compile_definitions(pose_demo_v1 PRIVATE DEMO_READ_DIAGNOSTIC=1)
    if(DEMO_D47_ISOLATION)
        target_compile_definitions(pose_demo_v1 PRIVATE DEMO_D47_ISOLATION=1)
    endif()
endif()
target_compile_options(pose_demo_v1 PRIVATE -Wall -Wextra -Wpedantic -ffp-contract=off)
target_link_options(pose_demo_v1 PRIVATE -Wl,--no-as-needed)
target_link_libraries(pose_demo_v1 PRIVATE Icraft::HostBackend Icraft::ZG330Backend Threads::Threads)
if(DEMO_D47_ISOLATION)
    if(NOT DEMO_READ_DIAGNOSTIC)
        message(FATAL_ERROR "Isolation requires the bounded diagnostic overlay")
    endif()
    add_executable(pose_static_read diagnostics/static_read.cpp ${B}/input_read_sdk.cpp ${B}/input_read_contract.cpp)
    target_include_directories(pose_static_read PRIVATE ${B})
    target_compile_options(pose_static_read PRIVATE -Wall -Wextra -Wpedantic -ffp-contract=off)
    target_link_options(pose_static_read PRIVATE -Wl,--no-as-needed)
    target_link_libraries(pose_static_read PRIVATE Icraft::HostBackend Icraft::ZG330Backend Threads::Threads)
endif()
