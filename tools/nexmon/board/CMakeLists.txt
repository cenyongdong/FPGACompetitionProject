cmake_minimum_required(VERSION 3.24)
project(nexmon_board_direct LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
option(NEXMON_BUILD_NPU "Build the separate unchanged pose180 SDK wrapper" OFF)
set(PROJECT_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../../..")
set(CSI_CORE "${PROJECT_ROOT}/software/pose180_pipeline/src/input/nexmon")
set(CSI_TESTS "${PROJECT_ROOT}/software/pose180_pipeline/tests/native/nexmon")
add_library(nexmon_ingress "${CSI_CORE}/ingress.cpp" "${CSI_CORE}/pcap_stream.cpp")
target_compile_options(nexmon_ingress PRIVATE -Wall -Wextra -Wpedantic -ffp-contract=off)
target_include_directories(nexmon_ingress PUBLIC "${CSI_CORE}" "${CMAKE_CURRENT_SOURCE_DIR}")
add_library(nexmon_adaptive "${CSI_CORE}/adaptive_ingress.cpp")
target_link_libraries(nexmon_adaptive PUBLIC nexmon_ingress)
target_compile_options(nexmon_adaptive PRIVATE -Wall -Wextra -Wpedantic -ffp-contract=off)
add_executable(nexmon_adaptive_test "${CSI_TESTS}/test_adaptive_ingress.cpp")
target_link_libraries(nexmon_adaptive_test PRIVATE nexmon_adaptive)
add_executable(nexmon_adaptive_replay adaptive_replay.cpp)
target_link_libraries(nexmon_adaptive_replay PRIVATE nexmon_adaptive)
add_executable(nexmon_replay replay.cpp)
target_link_libraries(nexmon_replay PRIVATE nexmon_ingress)
add_executable(nexmon_ingress_test "${CSI_TESTS}/test_ingress.cpp")
target_link_libraries(nexmon_ingress_test PRIVATE nexmon_ingress)
add_executable(nexmon_pcap_stream_test "${CSI_TESTS}/test_pcap_stream.cpp")
target_link_libraries(nexmon_pcap_stream_test PRIVATE nexmon_ingress)
add_executable(csi_direct_receiver direct_receiver.cpp "${PROJECT_ROOT}/demo_v1/shared_fp16/preprocess.cpp")
target_include_directories(csi_direct_receiver PRIVATE "${PROJECT_ROOT}/demo_v1/board" "${PROJECT_ROOT}/demo_v1/shared_fp16")
target_compile_options(csi_direct_receiver PRIVATE -Wall -Wextra -Wpedantic -ffp-contract=off)
target_link_libraries(csi_direct_receiver PRIVATE nexmon_ingress pthread)
if(NEXMON_BUILD_NPU)
    find_package(icraft-hostbackend 3.39.0.0 EXACT REQUIRED)
    find_package(icraft-zg330backend 3.39.0.0 EXACT REQUIRED)
    target_sources(csi_direct_receiver PRIVATE pose180_runtime.cpp)
    target_compile_definitions(csi_direct_receiver PRIVATE NEXMON_WITH_NPU=1)
    target_link_options(csi_direct_receiver PRIVATE -Wl,--no-as-needed)
    target_link_libraries(csi_direct_receiver PRIVATE Icraft::HostBackend Icraft::ZG330Backend)
endif()
