cmake_minimum_required(VERSION 3.24)
project(pose_application LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(Threads REQUIRED)
find_package(icraft-hostbackend 3.39.0.0 EXACT REQUIRED)
find_package(icraft-zg330backend 3.39.0.0 EXACT REQUIRED)
add_executable(pose_application_check application_check.cpp application_core.cpp application_bridge.cpp mixed_host_check.cpp host_cpu_adapter.cpp preprocess.cpp tcp_receiver.cpp)
target_include_directories(pose_application_check PRIVATE .)
target_link_libraries(pose_application_check PRIVATE Icraft::HostBackend Icraft::ZG330Backend Threads::Threads)
target_compile_options(pose_application_check PRIVATE -Wall -Wextra -Wpedantic -ffp-contract=off)
add_executable(pose_transport_check tcp_transport_check.cpp tcp_receiver.cpp)
target_include_directories(pose_transport_check PRIVATE .)
target_link_libraries(pose_transport_check PRIVATE Threads::Threads)
target_compile_options(pose_transport_check PRIVATE -Wall -Wextra -Wpedantic -ffp-contract=off)
