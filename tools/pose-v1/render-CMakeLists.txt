cmake_minimum_required(VERSION 3.24)
project(pose_render_application LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(Threads REQUIRED)
find_package(icraft-hostbackend 3.39.0.0 EXACT REQUIRED)
find_package(icraft-zg330backend 3.39.0.0 EXACT REQUIRED)
add_executable(pose_application_render_check application_render_check.cpp application_core.cpp application_bridge.cpp mixed_host_check.cpp host_cpu_adapter.cpp preprocess.cpp tcp_receiver.cpp skeleton_render.cpp)
target_include_directories(pose_application_render_check PRIVATE .)
target_link_libraries(pose_application_render_check PRIVATE Icraft::HostBackend Icraft::ZG330Backend Threads::Threads)
target_compile_options(pose_application_render_check PRIVATE -Wall -Wextra -Wpedantic -ffp-contract=off)
add_executable(pose_skeleton_render_check skeleton_render_check.cpp skeleton_render.cpp)
target_include_directories(pose_skeleton_render_check PRIVATE .)
target_compile_options(pose_skeleton_render_check PRIVATE -Wall -Wextra -Wpedantic -ffp-contract=off)
add_executable(pose_video_capability_probe video_capability_probe.cpp)
target_compile_options(pose_video_capability_probe PRIVATE -Wall -Wextra -Wpedantic)
