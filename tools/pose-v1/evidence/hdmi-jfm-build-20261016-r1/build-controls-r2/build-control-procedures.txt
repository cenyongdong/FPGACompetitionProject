PROCEDURE findFiles
ARGS basedir pattern


    # Fix the directory name, this ensures the directory name is in the
    # native format for the platform and contains a final directory seperator
    set basedir [string trimright [file join [file normalize $basedir] { }]]
    set fileList {}
    # Look in the current directory for matching files, -type {f r}
    # means ony readable normal files are looked at, -nocomplain stops
    # an error being thrown if the returned list is empty
    foreach fileName [glob -nocomplain -type {f r} -path $basedir $pattern] {
        lappend fileList $fileName
    }

    # Now look for any sub direcories in the current directory
    foreach dirName [glob -nocomplain -type {d  r} -path $basedir *] {
        # Recusively call the routine on the sub directory and append any
        # new files to the results
        set subDirList [findFiles $dirName $pattern]
        if { [llength $subDirList] > 0 } {
            foreach subDirFile $subDirList {
                lappend fileList $subDirFile
            }
        }
    }
    return $fileList

PROCEDURE get_change_speed_value

PROCEDURE getfiles
ARGS dir baseDir

    set allFiles [list]
    foreach item [glob -nocomplain -directory $dir *] {
        if {[file isdirectory $item]} {
            lappend allFiles {*}[getfiles $item $baseDir]
        } else {
			set relativePath [string range $item [string length $baseDir] end]
            lappend allFiles $relativePath
        }
    }
    return $allFiles

PROCEDURE runEco
ARGS type mode force

    set device_part [get_property PART [current_project]]
    if {$type==2} {
#eco user design
        set result [ecoBramSDP $mode]
        if {$result} {
            writeDCP
            puts "complete!"
        }
    } elseif {$type==0} {
#eco ip 
		if {[regexp {325t} $device_part] || [regexp {690t} $device_part]} {
		    ecoIP $mode $force
		    
		} else {
		    puts "The $device_part device don't need BRAM patch!"
		}
    } elseif {$type==1} {  
		if {[regexp {325t} $device_part] || [regexp {690t} $device_part]} {
		   checkSDP
		} else {
		    puts "The $device_part device don't need BRAM patch!"
		}
    } elseif {$type==3} {
	    if {[regexp {325t} $device_part] || [regexp {690t} $device_part]} {
		    ecoBramSDP $mode
		} else {
		    puts "The  $device_part device don't need BRAM patch!"  
		}
#       write_checkpoint -force ./test.dcp
#       incrPR
    } else {
        puts "Unsopported usage!"
        puts "###usage:###"
        puts "eco IP: runEco 0"
        puts "check whole design: runEco 1"
        puts "eco user design: runEco 2"
        puts "############"
    }

PROCEDURE source_file_backup
ARGS source_file_path

    set dst_file_path [format "%s.bak" $source_file_path]
    file copy -force $source_file_path $dst_file_path
