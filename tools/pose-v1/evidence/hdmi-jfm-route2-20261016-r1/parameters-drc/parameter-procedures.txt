PROCEDURE disable_icap_for_psoc

PROCEDURE post_write_bitstream_patch

	set str_list $args
	if {[check_current_platform] == "windows"} {
		puts "write_bitstream post hook tcl begins to run."
		
		set default_path [pwd]
		
		regsub {/\w+\.runs/\w+$} $default_path "" prj_directory
		set procise_incr_cfg_path [file join $prj_directory "procise_incr_cfg.txt"]
		
		if {[file exists $procise_incr_cfg_path]} {
			set procise_run_tcl_path [file join $prj_directory "procise_run.tcl"]
			
			if {[catch {set procise_path [file join $::env(FMSH_PROCISE_PATH)]}]} {
					error "An error occurs:\nFind procise_incr_cfg.txt in current project,but FMSH_PROCISE_PATH doesn't exist.\nPlease set the FMSH_PROCISE_PATH environment variable correctly!"
			}
			
			set call_procise_bat_path [file join $prj_directory "call_procise.bat"]
			
			set procise_patch_enable [procise_incr_cfg_file_check $procise_incr_cfg_path] 
			if {$procise_patch_enable == 1} {
				generate_call_procise_bat $procise_path $procise_run_tcl_path $call_procise_bat_path
				generate_procise_run_tcl $procise_incr_cfg_path $procise_run_tcl_path
				exec cmd /c $call_procise_bat_path
				puts "write_bitstream post hook tcl runs successfully."
			} else {
				puts "write_bitstream post hook tcl runs completely, but it does nothing because the Procise increment configuration is disabled." 
			}
		}
	} elseif {[check_current_platform] == "linux"} {
		puts "write_bitstream post hook tcl begins to run."
		
		set default_path [pwd]
		
		regsub {/\w+\.runs/\w+$} $default_path "" prj_directory
		set procise_incr_cfg_path [file join $prj_directory "procise_incr_cfg.txt"]
		
		if {[file exists $procise_incr_cfg_path]} {
			set procise_run_tcl_path [file join $prj_directory "procise_run.tcl"]
			set call_procise_sh_path [file join $prj_directory "call_procise.sh"]
			
			if {[catch {set procise_path [file join $::env(FMSH_PROCISE_PATH)]}]} {
				error "An error occurs:\nFind procise_incr_cfg.txt in current project,but FMSH_PROCISE_PATH doesn't exist.\nPlease set the FMSH_PROCISE_PATH environment variable correctly!"
			}
			
			set procise_patch_enable [procise_incr_cfg_file_check $procise_incr_cfg_path] 
			if {$procise_patch_enable == 1} {
				generate_call_procise_sh $procise_path $procise_run_tcl_path $call_procise_sh_path
				generate_procise_run_tcl $procise_incr_cfg_path $procise_run_tcl_path
				exec bash $call_procise_sh_path
				puts "write_bitstream post hook tcl runs successfully."
			} else {
				puts "write_bitstream post hook tcl runs completely, but it does nothing because the Procise increment configuration is disabled." 
			}
		}
	}
	set device_value [get_device_value $str_list]
	set device_part [get_property PART [current_project]]
	set disable_icap_value [get_disable_icap_value $str_list]
	if {$disable_icap_value == "TRUE" && [regexp {xc7z} $device_part] || $disable_icap_value == "TRUE" && [regexp {fmql} $device_part]} {
		if {[catch {set procise_path [file join $::env(FMSH_PROCISE_PATH)]}]} {
			fmsh_send_msg_critical_warning "FMSH-->An error occurs:\nPart:$device_part need -disable_icap bit patch!\nEnvironment FMSH_PROCISE_PATH doesn't exist.\nPlease set the FMSH_PROCISE_PATH environment variable correctly!"
		} else {
			disable_icap_for_psoc $device_value
		}
	}
	if {$device_value != "0"} {
		if {[catch {set procise_path [file join $::env(FMSH_PROCISE_PATH)]}] && [catch {set hwdt_path [file join $::env(FMSH_HWDT_PATH)]}]} {
			if {[string match -nocase FMK230T $device_value]} {
				error "FMSH--> An error occurs:\ndevice_value:$device_value.Environment FMSH_PROCISE_PATH doesn't exist.\nPlease set the FMSH_PROCISE_PATH environment variable correctly!"
			}
		}
		if {[catch {set hwdt_path [file join $::env(FMSH_HWDT_PATH)]}]} {
			if {[string match -nocase FMK300T8 $device_value] || [string match -nocase FMK300T16 $device_value] || [string match -nocase FMK400T16 $device_value] || [string match -nocase FMK700T36 $device_value] || [string match -nocase FMK700T80 $device_value]} {
				error "FMSH--> An error occurs:\ndevice_value:$device_value.Environment FMSH_HWDT_PATH doesn't exist.\nPlease set the FMSH_HWDT_PATH environment variable correctly!"
			}
		}
		generate_final_bit $device_value
	}
