PROCEDURE disable_icap
ARGS input_bit_path device_name

	set default_path [pwd]	
	regsub {/\w+\.runs/\w+$} $default_path "" prj_directory
    set procise_path [file join $::env(FMSH_PROCISE_PATH)]
    set bit_name [file rootname [file tail $input_bit_path]]
	set revise_device_name [format "%s_revise_device" $bit_name]
    set disable_icap_name  [format "%s_disable_icap" $bit_name]
	set revise_device_path [file join [file dirname $input_bit_path] "$revise_device_name.bit"]
    set disable_icap_path [file join [file dirname $input_bit_path] "$disable_icap_name.bit"]
	
    set procise_part $device_name
    set procise_run_tcl_path [file join $prj_directory "procise_run_disable_icap.tcl"]
	set fw [open $procise_run_tcl_path "w"]
	puts $fw "set_device $procise_part"
	if {[regexp {JFMQL15T485} $procise_part]} {
		puts $fw "revise_bit_idcode $input_bit_path $revise_device_path -device_name xc7z015clg485-2"
		puts $fw "increment_bit_cfg_v3  $revise_device_path $disable_icap_path -disable_icap"
	} else {
		puts $fw "increment_bit_cfg_v3  $input_bit_path $disable_icap_path -disable_icap"
	}
    close $fw
	if {[check_current_platform] == "windows"} {
		set call_procise_bat_path [file join $prj_directory "call_procise_disable_icap.bat"]
		generate_call_procise_bat $procise_path $procise_run_tcl_path $call_procise_bat_path
		exec cmd /c $call_procise_bat_path
	} elseif {[check_current_platform] == "linux"} {
		set call_procise_sh_path [file join $prj_directory "call_procise_disable_icap.sh"]
		generate_call_procise_sh $procise_path $procise_run_tcl_path $call_procise_sh_path
		exec bash $call_procise_sh_path
	}
	file delete -force  $revise_device_path

PROCEDURE get_finally_bit_location
ARGS 

    set bit_list [glob -nocomplain [pwd]/*.bit]
    set bit_create_time_old 0
    #acquire lastly bit in the impl run
    if {$bit_list!=""} {
        foreach bit $bit_list {
        	set bit_create_time_new [file mtime $bit]
            if {$bit_create_time_new>$bit_create_time_old} {
        	    set bit_create_time_old $bit_create_time_new
                set bit_final $bit
            } else {
        	    set bit_create_time_old $bit_create_time_old
                set $bit_final $bit_final
            }
        }
    } else {
        puts "There is no bit in impl run!"
    }
    return $bit_final

PROCEDURE get_prj_path
ARGS 

  set current_dictory_path [pwd]
  puts "current_dictory_path is $current_dictory_path"
  set path_length [string length $current_dictory_path]
  
  for {set i [expr $path_length - 1]} {$i >= 0} {incr i -1} {
     if {[string index $current_dictory_path $i] == "/"} {
        set loc_index $i
        break
     } 
  }
  puts "loc_index is $loc_index."
  
  if {$loc_index < 10} {
     error "Exception occurs!\nInfo: loc_index = $loc_index"
  }
  
  set dst_str [string range $current_dictory_path 0 [expr $loc_index-1]]
  puts "dst_str is $dst_str"
  
  set prj_path [format "%s.xpr" [string range $dst_str 0 end-5]]
  puts "prj_path is $prj_path"
  return $prj_path
