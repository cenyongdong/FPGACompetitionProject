PROCEDURE get_vvd_path

PROCEDURE check_current_platform

	global tcl_platform env
	if {$tcl_platform(platform) eq "windows"} {
		puts "The operating system is WINDOWS!"
		return windows
	}
	if {![catch {open "/proc/cpuinfo"} f]} {
		puts "\nThe operating system is LINUX!\n"
		close $f
		return linux
	}
	error "An error occurs:\nUnknown Operating System!"
