PROCEDURE connect_startupe_edif
ARGS 

    if {[get_cells -hier -filter {NAME =~ "*ram_fmsh_startupe2_logic"}] != ""} {
	    set startupe2_logic_cell "bram_fmsh_startupe2_logic"	
	} else {
	    set startupe2_logic_cell [get_cells -hier -filter {REF_NAME == "STARTUPE2" && PRIMITIVE_LEVEL == "LEAF"}]
	}
    
	if {[get_cells -hier -filter {NAME =~ "*msh_dna_check_logic"}] != ""} {
	    set dna_check_logic_cell [get_cells -hier -filter {NAME =~ "*msh_dna_check_logic"}]
		set dna_check_valid_pin [get_pins [format "%s/valid" $dna_check_logic_cell]]
	    set dna_check_valid_complete_pin [get_pins [format "%s/valid_complete" $dna_check_logic_cell]]
		set dna_check_clk_pin [get_pins [format "%s/clk" $dna_check_logic_cell]]
	} else {
	    set dna_check_logic_cell  [get_cells -hier -filter {REF_NAME == "DNA_PORT" && PRIMITIVE_LEVEL == "LEAF"} -quiet]
	}
	
	
	#create gnd
	set gnd_name [format "%s_gnd" $startupe2_logic_cell]
    create_cell -reference gnd $gnd_name  
	set gnd_pin [get_pins [format "%s/G" $gnd_name]]
	
	#reset lut2
	set lut2_reset [format "%s_lut2_reset"  $startupe2_logic_cell]
	create_cell -reference LUT2 $lut2_reset
	set_property INIT 4'h1 [get_cells $lut2_reset]  
	set lut2_reset_I0_pin [get_pins [format "%s/I0" $lut2_reset]]
	set lut2_reset_I1_pin [get_pins [format "%s/I1" $lut2_reset]]
	set lut2_reset_O_pin [get_pins [format "%s/O" $lut2_reset]]
	
	#valid lut2
	set lut2_valid [format "%s_lut2_valid"  $startupe2_logic_cell]
	create_cell -reference LUT2 $lut2_valid
	set_property INIT 4'hD [get_cells $lut2_valid]  
	set lut2_valid_I0_pin [get_pins [format "%s/I0" $lut2_valid]]
	set lut2_valid_I1_pin [get_pins [format "%s/I1" $lut2_valid]]
	set lut2_valid_O_pin [get_pins [format "%s/O" $lut2_valid]]
	
	#GSR_pin 

	set gnd_net [create_net [format "%s_gnd_net" $startupe2_logic_cell]]
	set lut2_valid_O_net [create_net [format "%s_lut2_valid_O_net" $startupe2_logic_cell]]
	
	#connect gnd to lut2 reset I0
	connectNet $gnd_net $gnd_pin
	connectNet $gnd_net $lut2_reset_I0_pin
	
	if {[get_cells -hier -filter {NAME =~ "*msh_dna_check_logic"}] != ""} {
	    set valid_net [format "%s_valid_net" $dna_check_logic_cell]
	    set valid_complete_net [format "%s_valid_complete_net" $dna_check_logic_cell]
		if {[get_nets $valid_net] != ""} {
		#connect 
	        connectNet $valid_net  $lut2_valid_I1_pin
		    connectNet $valid_complete_net $lut2_valid_I0_pin
		
		    connectNet $lut2_valid_O_net $lut2_valid_O_pin
		    connectNet $lut2_valid_O_net $lut2_reset_I1_pin
		} else {
		    set valid_net [create_net [format "%s_valid_net" $dna_check_logic_cell]] 
		    set valid_complete_net [create_net [format "%s_valid_complete_net" $dna_check_logic_cell]]
		
		    connectNet $valid_net  $lut2_valid_I1_pin 
		    connectNet $valid_net  $dna_check_valid_pin
		
		    connectNet $valid_complete_net $lut2_valid_I0_pin
		    connectNet $valid_complete_net $dna_check_valid_complete_pin
		
		    connectNet $lut2_valid_O_net $lut2_valid_O_pin
		    connectNet $lut2_valid_O_net $lut2_reset_I1_pin
		
		    set BUFG_list [get_cells -hier -filter {REF_NAME == "BUFG" && PRIMITIVE_LEVEL == "LEAF"} -quiet]
			set i 0
		    set BUFG ""
		    if {[llength $BUFG_list] == 0} {
		        error "An exception occurs:\nNo BUFG Find in current design!(No clock source for check logic)"
		    } else {
		        foreach bufg $BUFG_list {
					set PERIOD [get_property PERIOD [get_clocks -of_objects [get_cells $bufg]] -quiet] 
	
			        if {$PERIOD >= 10} {
					    set BUFG_O_pin_bak [get_pins [format "%s/O" $bufg]]
		                set BUFG_O_net_bak [get_nets -of [get_pins $BUFG_O_pin_bak]]
						if {$BUFG_O_net_bak == ""} {
						    continue
						}
						if {[get_property DONT_TOUCH $BUFG_O_net_bak] != 1 && [regexp {dbg_hub} $BUFG_O_net_bak] == 0} {
						
						    set fanout [get_property FLAT_PIN_COUNT [get_nets $BUFG_O_net_bak]]
							puts "Find bufg:$bufg,PERIOD:$PERIOD,fanout:$fanout,BUFG_O_net_bak:$BUFG_O_net_bak"
							if {$fanout >= $i} {
							    set BUFG $bufg
								set i $fanout
							}
							
						}

				    }
			
			    }
		    }
		    if {[get_cells $BUFG] == ""} {
		        error "An error occurs:\nInfo:No correct BUFG source find in project!(No correct clock source for check logic)"
		    }
			set BUFG_O_pin [get_pins [format "%s/O" $BUFG]]
		    set BUFG_O_net [get_nets -of [get_pins $BUFG_O_pin]]
			puts "Correct BUFG:$BUFG,BUFG_O_net:$BUFG_O_net"
		    connectNet $BUFG_O_net $dna_check_clk_pin
		    
	    }	
	} else {
	    set id_ready_pin [get_pins -hier -filter {NAME =~ "*dna_judge_top_inst/dna_ready"}]
		if {[get_pins $id_ready_pin] != ""} {
		    set get_fpga_version_cell [get_cells bram_get_fpga_version]
		   
			regexp {(\S+)/\S+/\S+/dna_ready} $id_ready_pin match top
	        set dna_check_valid_complete_pin [get_pins [format "%s/dna_ready" $top]]
			set dna_check_valid_pin  [get_pins -hier -filter {NAME =~ "*ram_get_fpga_version/dna_valid"}]
			set valid_net [create_net [format "%s_valid_net" $get_fpga_version_cell]] 
			
			set fpga_valid_pin_i_list [get_pins -hier -filter {NAME =~ "*ram_get_fpga_version/dna_value_i*"}]
	        set i 0
	        foreach fpga_pin $fpga_valid_pin_i_list {
	            set id_pin [get_pins [format "%s/dna_value\[%s\]" $top $i] ]    
		        if {[get_nets -of [get_pins $id_pin]] != ""} {
		            set net [get_nets -of [get_pins $id_pin]]
			        set flag [getDontTouchProperty $net]
			        connectNet $net $fpga_pin
			        restoreDontTouchProperty $flag $net
		        } else {
		            set net [create_net [format "%s_id_value_net_%s" $get_fpga_version_cell $i]]
			        connectNet $net $id_pin
			        connectNet $net $fpga_pin
		        }
				set i [expr $i + 1]
	        }	
		
		
		
		    connectNet $valid_net  $lut2_valid_I1_pin 
		    connectNet $valid_net  $dna_check_valid_pin
		
		    if {[get_nets -of [get_pins $dna_check_valid_complete_pin]] != ""} {
			    set valid_complete_net [get_nets -of [get_pins $dna_check_valid_complete_pin]]
			    connectNet $valid_complete_net $lut2_valid_I0_pin
			} else {
			    set valid_complete_net [create_net [format "%s_valid_complete_net" $get_fpga_version_cell]]  
				connectNet $valid_complete_net $lut2_valid_I0_pin
		        connectNet $valid_complete_net $dna_check_valid_complete_pin
			}

		
		    connectNet $lut2_valid_O_net $lut2_valid_O_pin
		    connectNet $lut2_valid_O_net $lut2_reset_I1_pin    
		
		} else {
		    connectNet $valid_net  $lut2_valid_I1_pin 
		    connectNet $valid_net  $dna_check_valid_pin
		
		    connectNet $valid_complete_net $lut2_valid_I0_pin
		    connectNet $valid_complete_net $dna_check_valid_complete_pin
		
		    connectNet $lut2_valid_O_net $lut2_valid_O_pin
		    connectNet $lut2_valid_O_net $lut2_reset_I1_pin     
		
		}
			
	    

	}
	
	#connect  GSR to lut2 O
	set GSR_pin [get_pins [format "%s/GSR" $startupe2_logic_cell]]
	if {[get_cells -hier -filter {NAME =~ "*ram_fmsh_startupe2_logic"}] != ""} {
	    
	    set GSR_net [create_net [format "%s_gsr_net" $startupe2_logic_cell]]
	    connectNet $GSR_net $GSR_pin
	    connectNet $GSR_net $lut2_reset_O_pin
	} else {
	    set net_gsr [get_nets -of [get_pins $GSR_pin]] 
		set GSR_net [create_net [format "%s_gsr_net" $startupe2_logic_cell]]
        if {$net_gsr == ""} {
	        connectNet $GSR_net $GSR_pin
	        connectNet $GSR_net $lut2_reset_O_pin    
		} else {
			set lut2_or [format "%s_lut2_or"  $startupe2_logic_cell]
	        create_cell -reference LUT2 $lut2_or
	        set_property INIT 4'hE [get_cells $lut2_or]  
	        set lut2_or_I0_pin [get_pins [format "%s/I0" $lut2_or]]
	        set lut2_or_I1_pin [get_pins [format "%s/I1" $lut2_or]]
	        set lut2_or_O_pin [get_pins [format "%s/O" $lut2_or]]  
			
			set flag [getDontTouchProperty [get_nets $net_gsr]]
			
			unroute $net_gsr $GSR_pin
			connectNet $net_gsr $lut2_or_I0_pin
			connectNet $GSR_net $lut2_or_I1_pin
			connectNet $GSR_net $lut2_reset_O_pin  
			set lut2_or_O_net [create_net [format "%s_lut2_or_O_net" $startupe2_logic_cell]]
			connectNet $lut2_or_O_net $GSR_pin
			connectNet $lut2_or_O_net $lut2_or_O_pin
			
			restoreDontTouchProperty $flag $net_gsr
		
		}		
	
	}

PROCEDURE generate_procise_run_tcl

  set procise_incr_cfg_list [split [read $procise_incr_cfg_pointer] "\n"]

  foreach line $procise_incr_cfg_list {
      if {[regexp {^\s*#} $line]} {
      } elseif {[regexp {^\s+$} $line]} {
      } elseif {[regexp {^\s*device_type} $line]} {
         regsub {^\s+} $line "" line
         regsub -all {\s+} $line " " line
         set line_list [split $line " "]
         set device_type [lindex $line_list 1]
         puts $procise_run_tcl_pointer "set_device $device_type"
      } elseif {[regexp {^\s*input_bit} $line]} {
         regsub {^\s+} $line "" line
         regsub -all {\s+} $line " " line
         set line_list [split $line " "]
         set input_bit_name [lindex $line_list 1]
#         regsub -all {\\} $input_bit_path "/" input_bit_path
         set input_bit_path [file join $prj_impl_dir $input_bit_name]
         puts $procise_run_tcl_pointer "set input_bit_path $input_bit_path"
      } elseif {[regexp {^\s*output_bit} $line]} {
         regsub {^\s+} $line "" line
         regsub -all {\s+} $line " " line
         set line_list [split $line " "]
         set output_bit_name [lindex $line_list 1]
         set output_bit_path [file join $prj_impl_dir $output_bit_name]
#         regsub -all {\\} $output_bit_path "/" output_bit_path
         puts $procise_run_tcl_pointer "set output_bit_path $output_bit_path"
      } elseif {[regexp {^\s*-\w+} $line]} {
         if {[regexp {^\s*-gt_cfg\s+} $line]} {
             regsub -all {\\} $line "/" line 
             regsub {^\s+} $line "" line
             regsub -all {\s+} $line " " line
             set line_list [split $line " "]
             set gt_cfg_name [lindex $line_list 1]
             set gt_cfg_path [file join $prj_dir $gt_cfg_name]
             set line "-gt_cfg $gt_cfg_path"
         }
         append procise_incr_cfg_tcl_cmd " $line"
      }
  }
  puts $procise_run_tcl_pointer $procise_incr_cfg_tcl_cmd
 
  close $procise_run_tcl_pointer

  puts "$procise_run_tcl_path is generated successfully."

PROCEDURE procise_incr_cfg_file_check
ARGS procise_incr_cfg_path

 
  set procise_incr_cfg_pointer [open $procise_incr_cfg_path "r"]
  set procise_incr_cfg_list [split [read $procise_incr_cfg_pointer] "\n"]
  close $procise_incr_cfg_pointer
 
  set device_type_en 0
  set input_bit_en 0
  set output_bit_en 0
  
  set procise_patch_enable 0
  foreach line $procise_incr_cfg_list {
      if {[regexp {^\s*#} $line]} {
      } elseif {[regexp {^\s+$} $line]} {
      } elseif {[regexp {^\s*device_type} $line]} {
        set device_type_en 1
      } elseif {[regexp {^\s*input_bit} $line]} {
        set input_bit_en 1
      } elseif {[regexp {^\s*output_bit} $line]} {
        set output_bit_en 1
      } elseif {[regexp {^\s*-\w+} $line]} {
      }
  }
  if {$device_type_en && $input_bit_en && $output_bit_en} {
    set procise_patch_enable 1 
  } else { 
    set procise_patch_enable 0 
  }
  return $procise_patch_enable
