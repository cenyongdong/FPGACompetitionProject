// Public XIR file-backed Params only. No XRT, Session, or device dependencies.
#include <icraft-xir/core/network.h>
#include <icraft-xir/ops/hard_op.h>
#include "mixed_fusion_baseline.hpp"
#include <filesystem>
#include <fstream>
#include <iostream>
#include <stdexcept>
namespace {
namespace xir=icraft::xir;
namespace fs=std::filesystem;
void require(bool ok,const char* message){if(!ok){throw std::runtime_error(message);}}
void save(const xir::Params& p,const fs::path& raw,const fs::path& out,
          const std::string& name,int64_t op,const char* role){
    const auto bytes=p.storageBytes();
    require(bytes<=65536,"Unexpected parameter size");
    require(!p.isSwap(),"Swap-backed parameter unsupported");
    uint64_t offset=0;
    if(bytes){
        require(p.hasData() && p.isLazyLoaded(),"Public mapped RAW data unavailable");
        require(fs::equivalent(fs::path(p.lazyFilePath().string()),raw),"RAW source differs");
        offset=p.lazyFileOffset();
        require(offset<=fs::file_size(raw) && bytes<=fs::file_size(raw)-offset,"RAW range invalid");
    }
    std::ofstream f(out/name,std::ios::binary);
    require(bool(f),"Cannot create fresh parameter output");
    if(bytes){auto data=p.data<char>();require(bool(data),"Null public parameter data");f.write(data.get(),bytes);}
    f.close();require(bool(f),"Parameter output incomplete");
    std::cout<<"{\"op\":"<<op<<",\"role\":\""<<role<<"\",\"value\":"<<p->v_id
             <<",\"bytes\":"<<bytes<<",\"file\":\""<<name<<"\",\"lazy_offset\":"<<offset
             <<",\"lazy_loaded\":"<<(p.isLazyLoaded()?"true":"false")<<"}\n";
}
void inspect(const fs::path& root,const fs::path& out,const char* kind){
    const auto json=root/(std::string("piw24_")+kind+".json");
    const auto raw=root/(std::string("piw24_")+kind+".raw");
    auto graph=xir::Network::CreateFromJsonFile(json.string());
    graph.lazyLoadParamsFromFile(raw.string(),true);
    if(std::string(kind)=="ZG"){
        const auto& members=pose_v1::fusion_baseline::members.at(9186);
        require(members.size()==19,"Frozen group differs");
        int64_t count=0;size_t weights=0;
        for(auto id:members){
            auto op=graph.getOpById(id).cast<xir::HardOp>();require(bool(op),"Non-HardOp in frozen group");
            count+=op->layer_count;
            size_t i=0;
            for(const auto& p:op->instr){save(p,raw,out,"ZG-"+std::to_string(id)+"-instr-"+std::to_string(i++)+".bin",id,"instr");}
            i=0;
            for(const auto& p:op->params){require(id==8507,"Unexpected group parameter");++weights;save(p,raw,out,"ZG-8507-weight-"+std::to_string(i++)+".bin",id,"weight");}
        }
        require(count==11 && weights==1,"Frozen layers or weights differ");
    }else{
        const auto id=std::string(kind)=="adapted"?3414:673;
        auto op=graph.getOpById(id);require(op->inputs.size()==2 && op->inputs[1].isParams(),"Soft Matmul parameter differs");
        save(op->inputs[1].cast<xir::Params>(),raw,out,std::string(kind)+"-weight.bin",id,"weight");
    }
}
}
int main(int argc,char**argv){try{
    require(argc==3,"Model directory and fresh output directory required");
    const fs::path out(argv[2]);require(!fs::exists(out),"Preserve prior output");fs::create_directory(out);
    for(auto kind:{"ZG","adapted","optimized"}){inspect(argv[1],out,kind);}
    std::cout<<"{\"status\":\"public_RAW_extracted\",\"Device_opened\":false,\"Session_created\":false,\"md5_checked\":true}\n";
    return 0;
}catch(const std::exception& e){std::cerr<<e.what()<<'\n';return 1;}}
