cmake_minimum_required(VERSION 3.24)
project(pose180 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(icraft-hostbackend 3.39.0.0 EXACT REQUIRED)
find_package(icraft-zg330backend 3.39.0.0 EXACT REQUIRED)
add_executable(pose180_validate main.cpp)
target_compile_options(pose180_validate PRIVATE -Wall -Wextra -Wpedantic -ffp-contract=off)
target_link_options(pose180_validate PRIVATE -Wl,--no-as-needed)
target_link_libraries(pose180_validate PRIVATE Icraft::HostBackend Icraft::ZG330Backend)
