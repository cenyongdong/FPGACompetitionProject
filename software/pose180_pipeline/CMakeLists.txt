cmake_minimum_required(VERSION 3.24)
project(pose180_pipeline LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(Threads REQUIRED)
get_filename_component(PROJECT_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../.." ABSOLUTE)
set(CORE "${PROJECT_ROOT}/software/core/display")
set(PIPE_DISPLAY "${PROJECT_ROOT}/FPGA/hdmi_candidate/diagnostics/native_rgb565_r53/display")
set(PREPROCESS "${PROJECT_ROOT}/demo_v1/shared_fp16")
include_directories(board "${PREPROCESS}" "${PROJECT_ROOT}")
add_library(pose_postprocess ../core/pose/postprocess.cpp)
target_include_directories(pose_postprocess PUBLIC ../core/pose)
add_executable(postprocess_tests resident/postprocess_tests.cpp)
target_link_libraries(postprocess_tests PRIVATE pose_postprocess)
add_executable(pose_replay resident/pose_replay.cpp)
target_link_libraries(pose_replay PRIVATE pose_postprocess)
add_executable(filter_benchmark resident/filter_benchmark.cpp)
target_link_libraries(filter_benchmark PRIVATE pose_postprocess)
add_executable(pl_adapter_tests resident/pl_adapter_tests.cpp)
target_link_libraries(pl_adapter_tests PRIVATE pose_postprocess)
add_executable(fixed_filter_vectors resident/fixed_filter_vectors.cpp)
target_link_libraries(fixed_filter_vectors PRIVATE pose_postprocess)
add_executable(geometry_vectors resident/geometry_vectors.cpp)
target_link_libraries(geometry_vectors PRIVATE pose_postprocess)
add_executable(buffer_tests board/buffer_tests.cpp)
target_link_libraries(buffer_tests PRIVATE Threads::Threads)
add_executable(renderer_tests board/renderer_tests.cpp "${CORE}/hdmi_renderer.cpp"
    "${PROJECT_ROOT}/FPGA/hdmi_candidate/diagnostics/native_rgb565_r47/joint/projection.cpp")
target_include_directories(renderer_tests PRIVATE "${CORE}")
add_executable(display_state_tests resident/display_state_tests.cpp "${CORE}/hdmi_renderer.cpp"
    "${PROJECT_ROOT}/FPGA/hdmi_candidate/diagnostics/native_rgb565_r47/joint/projection.cpp")
target_include_directories(display_state_tests PRIVATE "${CORE}" resident)
option(POSE180_BUILD_SDK "Build the ARM64 SDK owner" OFF)
option(POSE180_DEMO_PROBE_FREE "Use the uninstrumented HDMI bitstream (no diagnostic MMIO)" OFF)
option(POSE180_DEMO_PROBE_FREE_DRAIN "Use probe-free HDMI with the bounded functional startup consumer" OFF)
option(POSE180_DEMO_PL_FILTER "Bind the new probe-free-drain-filter-v1 hardware profile" OFF)
option(POSE180_DEMO_PL_POST "Bind probe-free-drain-post-v2 with geometry and batched filtering" OFF)
if(POSE180_DEMO_PL_POST AND NOT POSE180_DEMO_PL_FILTER)
    message(FATAL_ERROR "PL postprocessing requires PL filter support")
endif()
if(POSE180_DEMO_PL_FILTER AND NOT POSE180_DEMO_PROBE_FREE_DRAIN)
    message(FATAL_ERROR "PL filter requires the matching bounded-consumer hardware profile")
endif()
set(DEMO_SOURCES resident/main.cpp resident/io.cpp resident/journal.cpp resident/csi.cpp resident/dataset.cpp
    "${PROJECT_ROOT}/tools/nexmon/board/ingress.cpp" "${PROJECT_ROOT}/tools/nexmon/board/adaptive_ingress.cpp" "${PROJECT_ROOT}/tools/nexmon/board/pcap_stream.cpp"
    "${PREPROCESS}/preprocess.cpp")
if(UNIX)
    add_executable(journal_tests resident/journal_tests.cpp resident/journal.cpp)
    target_link_libraries(journal_tests PRIVATE Threads::Threads)
    add_executable(pose180_demo_mock ${DEMO_SOURCES})
    target_include_directories(pose180_demo_mock PRIVATE "${PROJECT_ROOT}/tools/nexmon/board")
    target_compile_options(pose180_demo_mock PRIVATE -Wall -Wextra -Wpedantic -ffp-contract=off)
    target_link_libraries(pose180_demo_mock PRIVATE Threads::Threads)
    target_link_libraries(pose180_demo_mock PRIVATE pose_postprocess)
endif()
if(POSE180_BUILD_SDK)
    find_package(icraft-hostbackend 3.39.0.0 EXACT REQUIRED)
    find_package(icraft-zg330backend 3.39.0.0 EXACT REQUIRED)
    add_executable(pose180_pipeline board/main.cpp board/runtime.cpp "${PREPROCESS}/preprocess.cpp"
        "${CORE}/hdmi_renderer.cpp"
        "${PROJECT_ROOT}/FPGA/hdmi_candidate/diagnostics/native_rgb565_r47/joint/projection.cpp"
        "${PIPE_DISPLAY}/hdmi_output.cpp")
    target_include_directories(pose180_pipeline BEFORE PRIVATE "${PIPE_DISPLAY}")
    target_include_directories(pose180_pipeline PRIVATE "${CORE}"
        "${PROJECT_ROOT}/demo_v1/board" "${PROJECT_ROOT}/FPGA/hdmi_candidate/software")
    target_compile_options(pose180_pipeline PRIVATE -Wall -Wextra -Wpedantic -ffp-contract=off)
    target_link_options(pose180_pipeline PRIVATE -Wl,--no-as-needed)
    target_link_libraries(pose180_pipeline PRIVATE Icraft::HostBackend Icraft::ZG330Backend Threads::Threads)
    add_executable(pose180_demo ${DEMO_SOURCES} board/runtime.cpp "${CORE}/hdmi_renderer.cpp"
        "${PROJECT_ROOT}/FPGA/hdmi_candidate/diagnostics/native_rgb565_r47/joint/projection.cpp"
        "${PIPE_DISPLAY}/hdmi_output.cpp")
    target_compile_definitions(pose180_demo PRIVATE POSE180_DEMO_SDK=1)
    if(POSE180_DEMO_PROBE_FREE OR POSE180_DEMO_PROBE_FREE_DRAIN)
        target_compile_definitions(pose180_demo PRIVATE POSE180_DEMO_PROBE_FREE=1)
    endif()
    if(POSE180_DEMO_PROBE_FREE_DRAIN)
        target_compile_definitions(pose180_demo PRIVATE POSE180_DEMO_PROBE_FREE_DRAIN=1)
    endif()
    if(POSE180_DEMO_PL_FILTER)
        target_compile_definitions(pose180_demo PRIVATE POSE180_DEMO_PL_FILTER=1)
    endif()
    if(POSE180_DEMO_PL_POST)
        target_compile_definitions(pose180_demo PRIVATE POSE180_DEMO_PL_POST=1)
    endif()
    target_include_directories(pose180_demo BEFORE PRIVATE "${PIPE_DISPLAY}")
    target_include_directories(pose180_demo PRIVATE "${CORE}" "${PROJECT_ROOT}/demo_v1/board"
        "${PROJECT_ROOT}/FPGA/hdmi_candidate/software" "${PROJECT_ROOT}/tools/nexmon/board")
    target_compile_options(pose180_demo PRIVATE -Wall -Wextra -Wpedantic -ffp-contract=off)
    target_link_options(pose180_demo PRIVATE -Wl,--no-as-needed)
    target_link_libraries(pose180_demo PRIVATE Icraft::HostBackend Icraft::ZG330Backend Threads::Threads)
    target_link_libraries(pose180_demo PRIVATE pose_postprocess)
endif()
