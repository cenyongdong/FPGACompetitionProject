cmake_minimum_required(VERSION 3.24)
project(pose180_pipeline LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(Threads REQUIRED)
get_filename_component(PROJECT_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../.." ABSOLUTE)
set(CORE "${CMAKE_CURRENT_SOURCE_DIR}/src/display")
set(PIPE_DISPLAY "${CORE}")
set(PREPROCESS "${CMAKE_CURRENT_SOURCE_DIR}/src/preprocess")
# All production C++ modules are maintained under src/. SDK stub headers are
# added only to the corresponding test target below.
include_directories(src/app src/common src/input src/input/nexmon src/inference
    src/pose src/display src/io src/diagnostics src/compat "${PREPROCESS}" "${PROJECT_ROOT}")
add_library(pose_postprocess src/pose/postprocess.cpp src/pose/single_person.cpp)
add_library(pose_phase_branch ../core/input/phase_branch.cpp "${PREPROCESS}/preprocess.cpp")
target_compile_options(pose_phase_branch PRIVATE -Wall -Wextra -Wpedantic -ffp-contract=off)
add_executable(phase_branch_tests tests/native/phase_branch_tests.cpp)
target_link_libraries(phase_branch_tests PRIVATE pose_phase_branch)
add_executable(phase_preprocess "${PROJECT_ROOT}/tools/pose180/phase_preprocess.cpp")
target_link_libraries(phase_preprocess PRIVATE pose_phase_branch)
target_include_directories(pose_postprocess PUBLIC src/pose)
add_executable(postprocess_tests tests/native/postprocess_tests.cpp)
target_link_libraries(postprocess_tests PRIVATE pose_postprocess)
add_executable(single_person_tests tests/native/single_person_tests.cpp)
target_link_libraries(single_person_tests PRIVATE pose_postprocess)
add_executable(pose_replay tools/native/pose_replay.cpp)
target_link_libraries(pose_replay PRIVATE pose_postprocess)
add_executable(filter_benchmark tools/native/filter_benchmark.cpp)
target_link_libraries(filter_benchmark PRIVATE pose_postprocess)
add_executable(pl_adapter_tests tests/native/pl_adapter_tests.cpp)
target_link_libraries(pl_adapter_tests PRIVATE pose_postprocess)
add_executable(fixed_filter_vectors tools/native/fixed_filter_vectors.cpp)
target_link_libraries(fixed_filter_vectors PRIVATE pose_postprocess)
add_executable(geometry_vectors tools/native/geometry_vectors.cpp)
target_link_libraries(geometry_vectors PRIVATE pose_postprocess)
add_executable(buffer_tests tests/native/buffer_tests.cpp)
target_link_libraries(buffer_tests PRIVATE Threads::Threads)
add_executable(renderer_tests tests/native/renderer_tests.cpp "${CORE}/hdmi_renderer.cpp"
    "${CORE}/projection.cpp")
target_include_directories(renderer_tests PRIVATE "${CORE}")
add_executable(display_state_tests tests/native/display_state_tests.cpp "${CORE}/hdmi_renderer.cpp"
    "${CORE}/projection.cpp")
target_include_directories(display_state_tests PRIVATE "${CORE}" src/display)
option(POSE180_BUILD_SDK "Build the ARM64 SDK owner" OFF)
option(POSE180_DEMO_PROBE_FREE "Use the uninstrumented HDMI bitstream (no diagnostic MMIO)" OFF)
option(POSE180_DEMO_PROBE_FREE_DRAIN "Use probe-free HDMI with the bounded functional startup consumer" OFF)
option(POSE180_DEMO_PL_FILTER "Bind the new probe-free-drain-filter-v1 hardware profile" OFF)
option(POSE180_DEMO_PL_POST "Bind probe-free-drain-post-v2 with geometry and batched filtering" OFF)
if(POSE180_DEMO_PL_POST AND NOT POSE180_DEMO_PL_FILTER)
    message(FATAL_ERROR "PL postprocessing requires PL filter support")
endif()
if(POSE180_DEMO_PL_FILTER AND NOT POSE180_DEMO_PROBE_FREE_DRAIN)
    message(FATAL_ERROR "PL filter requires the matching bounded-consumer hardware profile")
endif()
set(DEMO_SOURCES src/app/main.cpp src/io/io.cpp src/diagnostics/journal.cpp src/input/csi.cpp src/input/dataset.cpp
    "${PROJECT_ROOT}/software/pose180_pipeline/src/input/nexmon/ingress.cpp" "${PROJECT_ROOT}/software/pose180_pipeline/src/input/nexmon/adaptive_ingress.cpp" "${PROJECT_ROOT}/software/pose180_pipeline/src/input/nexmon/pcap_stream.cpp"
    "${PREPROCESS}/preprocess.cpp")
if(UNIX)
    add_executable(release_guard_tests tests/native/release_guard_tests.cpp)
    add_executable(hdmi_resource_tests tests/native/hdmi_resource_tests.cpp "${PIPE_DISPLAY}/hdmi_output.cpp")
    target_include_directories(hdmi_resource_tests BEFORE PRIVATE tests/sdk_stub "${PIPE_DISPLAY}" "${CORE}")
    target_link_libraries(hdmi_resource_tests PRIVATE Threads::Threads)
    add_executable(journal_tests tests/native/journal_tests.cpp src/diagnostics/journal.cpp)
    target_link_libraries(journal_tests PRIVATE Threads::Threads)
    add_executable(pose180_demo_mock ${DEMO_SOURCES})
    target_include_directories(pose180_demo_mock PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}/src/input/nexmon")
    target_compile_options(pose180_demo_mock PRIVATE -Wall -Wextra -Wpedantic -ffp-contract=off)
    target_link_libraries(pose180_demo_mock PRIVATE Threads::Threads)
    target_link_libraries(pose180_demo_mock PRIVATE pose_postprocess)
endif()
if(POSE180_BUILD_SDK)
    find_package(icraft-hostbackend 3.39.0.0 EXACT REQUIRED)
    find_package(icraft-zg330backend 3.39.0.0 EXACT REQUIRED)
    add_executable(pose180_pipeline src/app/strict_main.cpp src/inference/runtime.cpp "${PREPROCESS}/preprocess.cpp"
        "${CORE}/hdmi_renderer.cpp"
        "${CORE}/projection.cpp"
        "${PIPE_DISPLAY}/hdmi_output.cpp")
    target_include_directories(pose180_pipeline BEFORE PRIVATE "${PIPE_DISPLAY}")
    target_include_directories(pose180_pipeline PRIVATE "${CORE}"
        "${CMAKE_CURRENT_SOURCE_DIR}/src/compat")
    target_compile_options(pose180_pipeline PRIVATE -Wall -Wextra -Wpedantic -ffp-contract=off)
    target_link_options(pose180_pipeline PRIVATE -Wl,--no-as-needed)
    target_link_libraries(pose180_pipeline PRIVATE Icraft::HostBackend Icraft::ZG330Backend Threads::Threads)
    add_executable(pose180_demo ${DEMO_SOURCES} src/inference/runtime.cpp "${CORE}/hdmi_renderer.cpp"
        "${CORE}/projection.cpp"
        "${PIPE_DISPLAY}/hdmi_output.cpp")
    target_compile_definitions(pose180_demo PRIVATE POSE180_DEMO_SDK=1)
    if(POSE180_DEMO_PROBE_FREE OR POSE180_DEMO_PROBE_FREE_DRAIN)
        target_compile_definitions(pose180_demo PRIVATE POSE180_DEMO_PROBE_FREE=1)
    endif()
    if(POSE180_DEMO_PROBE_FREE_DRAIN)
        target_compile_definitions(pose180_demo PRIVATE POSE180_DEMO_PROBE_FREE_DRAIN=1)
    endif()
    if(POSE180_DEMO_PL_FILTER)
        target_compile_definitions(pose180_demo PRIVATE POSE180_DEMO_PL_FILTER=1)
    endif()
    if(POSE180_DEMO_PL_POST)
        target_compile_definitions(pose180_demo PRIVATE POSE180_DEMO_PL_POST=1)
    endif()
    target_include_directories(pose180_demo BEFORE PRIVATE "${PIPE_DISPLAY}")
    target_include_directories(pose180_demo PRIVATE "${CORE}" "${CMAKE_CURRENT_SOURCE_DIR}/src/compat" "${CMAKE_CURRENT_SOURCE_DIR}/src/input/nexmon")
    target_compile_options(pose180_demo PRIVATE -Wall -Wextra -Wpedantic -ffp-contract=off)
    target_link_options(pose180_demo PRIVATE -Wl,--no-as-needed)
    target_link_libraries(pose180_demo PRIVATE Icraft::HostBackend Icraft::ZG330Backend Threads::Threads)
    target_link_libraries(pose180_demo PRIVATE pose_postprocess)
endif()
