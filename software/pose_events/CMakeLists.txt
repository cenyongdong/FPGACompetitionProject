cmake_minimum_required(VERSION 3.16)
project(pose_events LANGUAGES CXX)
add_library(pose_events reference.cpp)
target_compile_features(pose_events PUBLIC cxx_std_17)
target_include_directories(pose_events PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
if(MSVC)
  target_compile_options(pose_events PRIVATE /W4 /fp:strict)
else()
  target_compile_options(pose_events PRIVATE -Wall -Wextra -Wpedantic -ffp-contract=off)
endif()
add_executable(pose_events_offline offline.cpp)
target_link_libraries(pose_events_offline PRIVATE pose_events)
include(CTest)
if(BUILD_TESTING)
  add_executable(pose_events_tests test/reference_tests.cpp)
  target_link_libraries(pose_events_tests PRIVATE pose_events)
  add_test(NAME pose_events_contract COMMAND pose_events_tests)
endif()
