cmake_minimum_required(VERSION 3.16)
project(pose_v1 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_library(pose_preprocess src/preprocess.cpp)
target_include_directories(pose_preprocess PUBLIC include)
if(MSVC)
    target_compile_options(pose_preprocess PRIVATE /W4)
else()
    target_compile_options(pose_preprocess PRIVATE -Wall -Wextra -Wpedantic)
endif()
if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
    target_compile_options(pose_preprocess PRIVATE -ffp-contract=off)
endif()
add_executable(pose_preprocess_check src/preprocess_check.cpp)
target_link_libraries(pose_preprocess_check PRIVATE pose_preprocess)
enable_testing()
add_test(NAME preprocess_self_check COMMAND pose_preprocess_check --self-test)

# OFF retains the existing SDK-independent preprocessing build.
option(POSE_BUILD_ICRAFT_CHECK "Build the independent Icraft inference checker" OFF)
option(POSE_ENABLE_ZG330 "Include the Lite hardware backend (not needed for Host reference)" ON)
if(POSE_BUILD_ICRAFT_CHECK)
    find_package(icraft-hostbackend CONFIG REQUIRED)
    add_executable(pose_inference_check src/inference_check.cpp)
    target_link_libraries(pose_inference_check PRIVATE pose_preprocess Icraft::HostBackend)
    if(POSE_ENABLE_ZG330)
        find_package(icraft-zg330backend CONFIG REQUIRED)
        target_link_libraries(pose_inference_check PRIVATE Icraft::ZG330Backend)
        target_compile_definitions(pose_inference_check PRIVATE POSE_ENABLE_ZG330=1)
    endif()
endif()
