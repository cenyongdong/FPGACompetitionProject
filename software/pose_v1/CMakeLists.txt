cmake_minimum_required(VERSION 3.16)
project(pose_v1 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_library(pose_preprocess src/preprocess.cpp)
target_include_directories(pose_preprocess PUBLIC include)
target_compile_options(pose_preprocess PRIVATE -Wall -Wextra -Wpedantic)
if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
    target_compile_options(pose_preprocess PRIVATE -ffp-contract=off)
endif()
add_executable(pose_preprocess_check src/preprocess_check.cpp)
target_link_libraries(pose_preprocess_check PRIVATE pose_preprocess)
enable_testing()
add_test(NAME preprocess_self_check COMMAND pose_preprocess_check --self-test)
